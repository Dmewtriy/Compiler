Грамматика

G[Z] = {Vt, Vn, Z, P}
Vt = {'enum', 'case', '{', '}', ';', 'space', 'a'...'z', 'A'...'Z', ‘0’…’9’}
Vn = {Z, ENUM, SPACE, ID, ENUM_BODY, CASE, SPACE_BTWN, CASE_ID, END_CASE, END_BODY, END, Б, Ц}

P = {
    Z -> 'enum' ENUM
    ENUM -> 'space' SPACE
    SPACE -> ('_' | Б){Б | Ц | ‘_’} ID
    ID-> '{' ENUM_BODY
    ENUM_BODY -> 'case' CASE | '}' END_BODY
    CASE-> 'space' SPACE_BTWN
    SPACE_BTWN-> ('_' | Б){Б | Ц | ‘_’} CASE_ID
    CASE_ID-> ';' END_CASE
    END_CASE -> 'case' CASE | '}' END_BODY
    END_BODY -> ';' END
    Б → ‘a’ | ‘b’ | … | ‘z’ | ‘A’ | ‘B’ | … | ‘Z’
    Ц → ‘0’ | ‘1’ | … | ‘9’
}
